      <h2>Get Started in 5 Minutes</h2>
      <p>Create your first AMCP agent with Quarkus and Kafka</p>
    </div>

    <div class="quickstart-steps">
      <div class="step">
        <div class="step-number">1</div>
        <h3>Create Quarkus Project</h3>
        <pre><code>quarkus create app my-agent-app \
  --extension=amcp-quarkus</code></pre>
      </div>

      <div class="step">
        <div class="step-number">2</div>
        <h3>Add Kafka Extension</h3>
        <pre><code>quarkus extension add kafka</code></pre>
      </div>

      <div class="step">
        <div class="step-number">3</div>
        <h3>Create Your Agent</h3>
        <pre><code>@QuarkusMain
public class MyAgent extends Agent {
  @Override
  public void initialize(AgentContext ctx) {
    ctx.subscribe("events", this::handle);
  }
}</code></pre>
      </div>

      <div class="step">
        <div class="step-number">4</div>
        <h3>Run in Dev Mode</h3>
        <pre><code>quarkus dev</code></pre>
      </div>
    </div>

    <div class="quickstart-cta">
